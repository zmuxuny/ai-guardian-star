 return {'total':len(rows),'valid_results':len(valid),'request_errors':len(rows)-len(valid),'correct':sum(x['intent']==x['expected'] for x in valid),'help_total':sum(x['expected']=='needs_help' for x in rows),'help_missed':sum(x['expected']=='needs_help' and (x['status']!='ok' or x.get('intent')!='needs_help') for x in rows),'wrong_okay':sum(x['intent']=='okay' and x['expected']!='okay' for x in valid),'false_help':sum(x['intent']=='needs_help' and x['expected']!='needs_help' for x in valid),'unclear_on_definite':sum(x['intent']=='unclear' and x['expected']!='unclear' for x in valid),'error_types':dict(collections.Counter(x['error'] for x in rows if x['status']!='ok')),'confusion':{label:dict(collections.Counter(x.get('intent','error') if x['status']=='ok' else 'error' for x in rows if x['expected']==label)) for label in ['okay','needs_help','unclear']}}
